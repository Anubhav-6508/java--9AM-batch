	- Read numbers from end-user from command line 
	- divide them and display result

class Test03 {
	public static void main(String[] args) {

		int a = Integer.parseInt(args[0]);
		int b = Integer.parseInt(args[1]);
		int c = a/b ;
		System.out.println("Result: "+ c);

	}
}
